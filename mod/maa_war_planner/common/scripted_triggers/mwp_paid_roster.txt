mwp_paid_slot_current = {
 trigger_if = { limit = { exists = var:mwp_paid_unit_$N$ }
 var:mwp_paid_unit_$N$ = {
    regiment_owner = root is_personal_maa_regiment = yes is_regular_maa_regiment = yes
    is_raised = no is_maa_in_combat = no
    maa_size = root.var:mwp_paid_size_$N$ maa_max_troops_count = root.var:mwp_paid_cap_$N$
    maa_current_troops_count = root.var:mwp_paid_soldiers_$N$ mwp_florin_hired_type_id = root.var:mwp_paid_type_$N$
    trigger_if = { limit = { root = { has_variable = mwp_paid_expand_$N$ } } can_upgrade_maa = yes }
    trigger_else = { can_upgrade_maa = no }
    trigger_if = { limit = { root = { has_variable = mwp_paid_station_$N$ } } exists = regiment_station regiment_station = root.var:mwp_paid_station_$N$ }
    trigger_else = { NOT = { exists = regiment_station } }
 }
 }
 trigger_else = { always = no }
}
