mwp_d21_nomad_unraised_trigger = {
 government_has_flag = government_is_nomadic exists = domicile NOT = { has_raised_armies = yes }
}
mwp_d21_nomad_selected_resources_trigger = {
 trigger_if = { limit = { mwp_d21_nomad_mode_value = 1 }
  mwp_d21_active_route_value = 1 mwp_d21_nomad_unraised_trigger = yes
  domicile.herd >= mwp_d21_nomad_herd_cost_value domicile.horde >= mwp_d21_nomad_herd_cost_value
  trigger_if = { limit = { has_variable = mwp_d19_tx_active }
   exists = var:mwp_d21_tx_domicile domicile = var:mwp_d21_tx_domicile
   domicile = { herd = root.var:mwp_d21_tx_herd_before }
  }
 }
}
