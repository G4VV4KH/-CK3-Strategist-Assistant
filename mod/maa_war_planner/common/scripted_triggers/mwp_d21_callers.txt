mwp_d21_type_budget_1_trigger = {
    trigger_if = {
        limit = {
            mwp_d21_upkeep_free_trigger = yes
        }
        always = yes
    }
    trigger_else_if = {
        limit = {
            mwp_auto_budget_mode_value = 3
        }
        always = yes
    }
    trigger_else = {
        trigger_if = {
            limit = {

            }
            mwp_d21_upkeep_bound_1 >= 0
            trigger_if = {
                limit = {
                    mwp_d21_upkeep_currency_1 = 1
                    mwp_d21_legacy_gold_budget_trigger = yes
                }
                var:mwp_auto_B >= mwp_d21_upkeep_bound_1
            }
            trigger_else = {
                mwp_d21_budget_ledger_current_trigger = yes
                trigger_if = {
                    limit = {
                        mwp_auto_budget_mode_value = 1
                    }
                    mwp_d21_whole_army_other_wallets_trigger = yes
                }
                OR = {
                    AND = {
                        mwp_d21_upkeep_currency_1 = 1
                        has_variable = mwp_d21b_headroom_1
                        var:mwp_d21b_headroom_1 >= mwp_d21_upkeep_bound_1
                    }
                    AND = {
                        mwp_d21_upkeep_currency_1 = 2
                        has_variable = mwp_d21b_headroom_2
                        var:mwp_d21b_headroom_2 >= mwp_d21_upkeep_bound_1
                    }
                    AND = {
                        mwp_d21_upkeep_currency_1 = 3
                        has_variable = mwp_d21b_headroom_3
                        var:mwp_d21b_headroom_3 >= mwp_d21_upkeep_bound_1
                    }
                    AND = {
                        mwp_d21_upkeep_currency_1 = 4
                        has_variable = mwp_d21b_headroom_4
                        var:mwp_d21b_headroom_4 >= mwp_d21_upkeep_bound_1
                    }
                    AND = {
                        mwp_d21_upkeep_currency_1 = 5
                        has_variable = mwp_d21b_headroom_5
                        var:mwp_d21b_headroom_5 >= mwp_d21_upkeep_bound_1
                    }
                }
            }
        }
        trigger_else = {
            always = no
        }
    }
}
