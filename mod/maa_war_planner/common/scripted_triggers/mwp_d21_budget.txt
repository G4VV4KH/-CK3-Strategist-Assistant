  has_variable = mwp_d21b_day has_variable = mwp_d21b_mode

  has_variable = mwp_d21b_stamp has_variable = mwp_d21r_clock
  has_variable = mwp_d21b_counter
 }
  var:mwp_d21b_ready = yes var:mwp_d21b_owner = this
  var:mwp_d21b_generation = var:mwp_generation
  var:mwp_d21b_day = days_since_birth var:mwp_d21b_mode = mwp_auto_budget_mode_value
  var:mwp_d21b_epoch = var:mwp_d21r_epoch

 }

}

trigger_if = { limit = { OR = { mwp_auto_budget_mode_value = 3 mwp_d21_upkeep_free_trigger = yes } } always = yes }
 trigger_else = {
  mwp_d21_budget_ledger_identity_trigger = yes
  mwp_native_cost_purchase_count_value = mwp_d21_budget_expected_counter_value
  has_variable = mwp_script_selected_upkeep_currency has_variable = mwp_script_selected_upkeep
  OR = {
AND = {
var:mwp_script_selected_upkeep_currency = 1
 var:mwp_d21b_headroom_1 >= var:mwp_script_selected_upkeep
 var:mwp_d21b_balance_1 >= { value = var:mwp_d21b_reserved_1 add = var:mwp_script_selected_upkeep add = 0.01 }
monthly_character_expenses <= { value = var:mwp_d21b_income_1 subtract = 0.01 }

}
AND = {

monthly_character_men_at_arms_expense_prestige <= { value = var:mwp_d21b_initial_prestige_maa add = var:mwp_d21b_reserved_2 add = var:mwp_script_selected_upkeep }

mwp_d21_candidate_budget_trigger = {

 trigger_else = {
 trigger_if = { limit = { has_variable = mwp_script_candidate_upkeep_currency has_variable = mwp_script_candidate_upkeep }
  var:mwp_script_candidate_upkeep >= 0
  trigger_if = { limit = { var:mwp_script_candidate_upkeep_currency = 1 mwp_d21_legacy_gold_budget_trigger = yes }
   var:mwp_auto_B >= var:mwp_script_candidate_upkeep
  }
  trigger_else = {
   mwp_d21_budget_ledger_current_trigger = yes
   trigger_if = { limit = { mwp_auto_budget_mode_value = 1 } mwp_d21_whole_army_other_wallets_trigger = yes }

    AND = { var:mwp_script_candidate_upkeep_currency = 1

     var:mwp_d21b_headroom_1 >= var:mwp_script_candidate_upkeep
