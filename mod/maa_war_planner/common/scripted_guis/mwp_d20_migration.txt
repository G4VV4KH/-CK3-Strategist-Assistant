# Current schema gates paid freshness independently of polling order.
mwp_d20_migrate_wars = { scope = character
 is_valid = { is_ai = no is_alive = yes
 NOT = { mwp_d20_war_schema_current_trigger = yes }
 NOT = { AND = { has_variable = mwp_analysis_requested var:mwp_analysis_requested = yes } }
 NOT = { AND = {
  has_variable = mwp_d20_migration_owner var:mwp_d20_migration_owner = this
  has_variable = mwp_d20_migration_generation has_variable = mwp_generation
  var:mwp_d20_migration_generation = var:mwp_generation
 } } }
 effect = { if = { limit = { is_ai = no is_alive = yes
 NOT = { mwp_d20_war_schema_current_trigger = yes }
 NOT = { AND = { has_variable = mwp_analysis_requested var:mwp_analysis_requested = yes } }
 NOT = { AND = {
  has_variable = mwp_d20_migration_owner var:mwp_d20_migration_owner = this
  has_variable = mwp_d20_migration_generation has_variable = mwp_generation
  var:mwp_d20_migration_generation = var:mwp_generation
 } } }
  mwp_request_analysis_effect = yes
  set_variable = { name = mwp_d20_migration_owner value = this }
  set_variable = { name = mwp_d20_migration_generation value = var:mwp_generation }
 } }
}
