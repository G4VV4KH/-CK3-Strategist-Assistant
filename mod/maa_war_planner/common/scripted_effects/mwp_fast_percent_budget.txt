    remove_variable = mwp_d19_net_ready
    remove_variable = mwp_d19_net_owner
    remove_variable = mwp_d19_net_generation
    remove_variable = mwp_d19_net_day
    remove_variable = mwp_d19_net_mode
    remove_variable = mwp_d19_net_gross
    remove_variable = mwp_d19_net_income
    remove_variable = mwp_d19_net_expense_limit
    remove_variable = mwp_d19_net_initial_expenses
    remove_variable = mwp_d19_net_initial_balance
    remove_variable = mwp_d19_net_purchase_count
    if = {
        limit = {
            NOT = {
                mwp_auto_budget_mode_value = 3
            }
        }
        if = {
            limit = { mwp_fbr_live = yes has_variable = mwp_script_budget_ready has_variable = mwp_auto_B }
            if = { limit = { var:mwp_script_budget_ready = yes var:mwp_script_budget_income = scope:mwp_fbr_gross }
                set_variable = {
                    name = mwp_d19_net_owner
                    value = this
                }
                set_variable = {
                    name = mwp_d19_net_generation
                    value = var:mwp_generation
                }
                set_variable = {
                    name = mwp_d19_net_day
                    value = days_since_birth
                }
                set_variable = {
                    name = mwp_d19_net_mode
                    value = mwp_auto_budget_mode_value
                }
                set_variable = {
                    name = mwp_d19_net_gross
                    value = scope:mwp_fbr_gross
                }
                set_variable = {
                    name = mwp_d19_net_initial_expenses
                    value = { value = scope:mwp_fbr_raw multiply = -1 }
                }
                set_variable = {
                    name = mwp_d19_net_initial_balance
                    value = { value = scope:mwp_fbr_net max = scope:mwp_input_current_balance }
                }
                set_variable = {
                    name = mwp_d19_net_purchase_count
                    value = mwp_native_cost_purchase_count_value
                }
                set_variable = {
                    name = mwp_d19_net_income
                    value = {
                        value = { value = scope:mwp_fbr_net max = scope:mwp_input_current_balance }
                        add = { value = scope:mwp_fbr_raw multiply = -1 }
                        max = scope:mwp_fbr_gross
                    }
                }
                set_variable = {
                    name = mwp_d19_net_expense_limit
                    value = {
                        value = var:mwp_d19_net_income
                        subtract = 0.01
                    }
                }
                set_variable = {
                    name = mwp_auto_B
                    value = {
                        value = var:mwp_d19_net_expense_limit
                        subtract = monthly_character_expenses
                        max = var:mwp_auto_B
                    }
                }
                set_variable = {
                    name = mwp_d19_net_ready
                    value = yes
                }
            }
        }
        if = {
            limit = {
                NOT = {
                    has_variable = mwp_d19_net_ready
                }
            }
            set_variable = {
                name = mwp_auto_B
                value = -1
            }
            remove_variable = mwp_script_budget_ready
            set_variable = {
                name = mwp_script_budget_error
                value = 1
            }
        }
    }
}
