mwp_d21_pending_rollback_effect = {
    if = {
        limit = {
            mwp_d21_pending_recovery_matches_trigger = yes
        }
        if = {
            limit = {
                mwp_d21_nomad_mode_value = 1
                NOT = {
                    has_variable = mwp_d21p_herd_target
                }
            }
            set_variable = {
                name = mwp_d21p_herd_before
                value = domicile.herd
            }
            set_variable = {
                name = mwp_d21p_herd_target
                value = {
                    value = domicile.herd
                    add = var:mwp_d21_tx_herd_cost
                }
            }
        }
        if = {
            limit = {
                NOT = {
                    has_variable = mwp_d21p_refund_target
                }
            }
            set_variable = {
                name = mwp_d21p_refund_before
                value = mwp_d21_selected_wallet_value
            }
            set_variable = {
                name = mwp_d21p_refund_target
                value = {
                    value = mwp_d21_selected_wallet_value
                    add = var:mwp_d19_tx_price
                }
            }
        }
        if = {
            limit = {
                var:mwp_d19_tx_action = 2
            }
            var:mwp_d19_tx_regiment = {
                save_scope_value_as = {
                    name = mwp_d21p_current_before_inverse
                    value = {
                        value = maa_current_troops_count
                        max = root.var:mwp_d19_tx_max
                    }
                }
            }
            set_variable = {
                name = mwp_d21p_rollback_current
                value = scope:mwp_d21p_current_before_inverse
            }
        }
        if = {
            limit = {
                var:mwp_d19_tx_action = 2
            }
            var:mwp_d19_tx_regiment = {
                change_maa_regiment_size = {
                    size = -1
                    reinforce = no
                }
                if = {
                    limit = {
                        root = {
                            mwp_d21_nomad_mode_value = 1
                        }
                    }
                    change_maa_troops_count = {
                        value = root.var:mwp_d21p_rollback_current
                        subtract = maa_current_troops_count
                    }
                }
            }
        }
        else = {
            var:mwp_d19_tx_regiment = {
                destroy_maa_regiment = yes
            }
        }
        if = {
            limit = {
                mwp_d21_pending_restored_trigger = yes
                mwp_d21_selected_wallet_value = var:mwp_d21p_refund_before
                var:mwp_auto_purchase_count = var:mwp_d19_tx_expected_counter
                NOT = {
                    has_variable = mwp_d19_tx_refunded
                }
            }
            mwp_d21_refund_transaction_effect = yes
            mwp_d21_pending_refund_herd_effect = yes
            set_variable = {
                name = mwp_d19_tx_refunded
                value = yes
            }
            if = {
                limit = {
                    mwp_d21_selected_wallet_value = var:mwp_d21p_refund_target
                    trigger_if = {
                        limit = {
                            mwp_d21_nomad_mode_value = 1
                        }
                        domicile = {
                            herd = root.var:mwp_d21p_herd_target
                        }
                    }
                }
                set_variable = {
                    name = mwp_auto_purchase_count
                    value = var:mwp_d19_tx_counter
                }
                remove_variable = mwp_auto_last_paid_verified
                remove_variable = mwp_script_purchase_verified
                remove_variable = mwp_script_mutated_regiment
                remove_variable = mwp_batch_continue
                set_variable = {
                    name = mwp_d19_rollback_id
                    value = var:mwp_d19_tx_id
                }
                set_variable = {
                    name = mwp_d19_rollback_owner
                    value = this
                }
                set_variable = {
                    name = mwp_d19_rollback_day
                    value = var:mwp_d19_tx_day
                }
                set_variable = {
                    name = mwp_d19_rollback_gold
                    value = var:mwp_d21p_refund_target
                }
                set_variable = {
                    name = mwp_d19_rollback_expenses
                    value = var:mwp_d19_tx_expenses
                }
                set_variable = {
                    name = mwp_d19_rollback_unraised
                    value = var:mwp_d19_tx_unraised
                }
                set_variable = {
                    name = mwp_d19_transaction_status
                    value = 2
                }
                set_variable = {
                    name = mwp_d19_transaction_rebuild_requested
                    value = yes
                }
                set_variable = {
                    name = mwp_auto_status
                    value = 13
                }
                if = {
                    limit = {
                        mwp_d21_selected_legacy_rejection_trigger = yes
                    }
                    mwp_d19_reject_record_effect = yes
                }
                else = {
                    mwp_d21_reject_record_effect = yes
                }
                mwp_native_cost_invalidate_effect = yes
                mwp_d19_transaction_clear_effect = yes
            }
            else = {
                mwp_d19_transaction_failure_effect = yes
            }
        }
        else = {
            mwp_d19_transaction_failure_effect = yes
        }
    }
    else = {
        mwp_d19_transaction_failure_effect = yes
    }
}
mwp_d21_pending_refund_herd_effect = {
 if = { limit = { mwp_d21_nomad_mode_value = 1 mwp_d21_pending_herd_owned_trigger = yes
  has_variable = mwp_d21p_herd_before has_variable = mwp_d21p_herd_target
  domicile = { herd >= root.var:mwp_d21p_herd_before herd <= root.var:mwp_d21p_herd_before }
 }
  domicile = { change_herd = root.var:mwp_d21_tx_herd_cost }
  set_variable = { name = mwp_d21_tx_herd_refunded value = yes }
 }
}
