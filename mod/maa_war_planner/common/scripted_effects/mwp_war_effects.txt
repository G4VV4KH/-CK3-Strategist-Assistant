# MWP war pool, CK3 1.20.1.2. Lists and scores belong to the player.
# Rivals/provinces store owner-tagged scan stamps for constant-time deduplication.
# API evidence: generated effects.log every_ruler:1057, every_war_enemy:5385,
# every_maa_regiment:4757, every_sub_realm_barony:5173; triggers.log
# is_unit_type:9382, maa_current_troops_count:9388, fort_level:10727.
# Installed examples: common/script_values/07_ep3_values.txt:4513;
# common/scripted_effects/00_ep1_inspiration_effects_sean.txt:540.
#
# Coverage: every count+ ruler supplied by every_ruler, including vassals, with
# native diplomatic-range and attack-hierarchy gates before CB queries, plus
# current war opponents. No neighbor-only or top-N cap. Landless coverage
# depends on the engine's ruler iterator and must be measured in a native test.
# has_any_cb_on excludes CBs disabled for interface availability; factions and
# scripted invasions without a direct active enemy are outside this prototype.
# Both broad directional pools are retained. mwp_war_scoring_rivals is the
# deduplicated union of shown+usable native war interactions and opposing active
# primary war actors; the script cache adds opposing coalition participants.
# This still is NOT selected CB/title CanSend; hook send-option coverage is pending.
# No manual exclusion for breakable truces/alliances is added. All aggregates
# and fort samples below use ONLY mwp_war_scoring_rivals.
# Legacy personal-regiment/strength summaries retired: no runtime consumers.
# Production collects native own owner/origin/title stats and coarse base enemy stats,
# actual active-war coalitions and conservative possible enemy allied support.
# DEV11 forts cover the scoring-rival realm union; capitals/borders are marked subsets.
# Realm provinces are geographic opportunities, NOT confirmed CB war objectives.

# Caller: player. Public refresh entry point.
mwp_war_refresh_effect = {
    save_scope_as = mwp_player
    mwp_war_reset_effect = yes
    mwp_read_analysis_strength_effect = {
        OUT = mwp_scan_player_strength
    }
    every_war_enemy = {
        limit = {
            is_alive = yes
            this != scope:mwp_player
        }
        save_scope_as = mwp_war_rival
        mwp_war_record_active_effect = yes
    }
    every_character_war = {
        if = {
            limit = {
                is_attacker = scope:mwp_player
            }
            primary_defender = {
                save_scope_as = mwp_war_rival
                mwp_war_record_scoring_union_effect = yes
            }
        }
        else = {
            primary_attacker = {
                save_scope_as = mwp_war_rival
                mwp_war_record_scoring_union_effect = yes
            }
        }
    }
    every_ruler = {
        limit = {
            is_alive = yes
            this != scope:mwp_player
        }
        save_scope_as = mwp_war_rival
        scope:mwp_player = {
            change_variable = {
                name = mwp_war_scanned_ruler_count
                add = 1
            }
        }
        if = {
            limit = {
                mwp_war_offensive_candidate_trigger = yes
            }
            mwp_scan_record_offensive_effect = yes
        }
        if = {
            limit = {
                mwp_war_defensive_candidate_trigger = yes
            }
            mwp_scan_record_defensive_effect = yes
        }
    }
    mwp_d20_rank_wars_effect = yes
    mwp_war_aggregate_effect = yes
    set_variable = {
        name = mwp_war_pool_schema
        value = 2
    }
    if = {
        limit = {
            var:mwp_war_terrain_unknown_count = 0
        }
        set_variable = {
            name = mwp_war_analysis_ready
            value = yes
        }
    }
    set_variable = {
        name = mwp_war_likelihood_ready
        value = yes
    }
}

# Caller: rival. Native-valid/active union used by all scoring and aggregates.
mwp_war_record_scoring_union_effect = {
    if = {
    limit = {
    NOT = { AND = { has_variable = mwp_war_scoring_owner var:mwp_war_scoring_owner = scope:mwp_player has_variable = mwp_war_scoring_stamp var:mwp_war_scoring_stamp = scope:mwp_player.var:mwp_war_scan_stamp } }
}

    set_variable = { name = mwp_war_scoring_owner value = scope:mwp_player }
    set_variable = { name = mwp_war_scoring_stamp value = scope:mwp_player.var:mwp_war_scan_stamp }
    scope:mwp_player = {
    add_to_variable_list = { name = mwp_war_scoring_rivals target = scope:mwp_war_rival }
    change_variable = { name = mwp_war_scoring_rival_count add = 1 }
}

}

# Caller: player. Each unique native-valid/active rival is evaluated once.
mwp_war_aggregate_effect = {
    if = {
        limit = {
            has_variable_list = mwp_war_scoring_rivals
        }
        every_in_list = {
            variable = mwp_war_scoring_rivals
            save_scope_as = mwp_war_rival
            if = {
                limit = {
                    exists = capital_province
                }
                capital_province = {
                    mwp_war_mark_region_capital_effect = yes
                }
            }
            if = {
                limit = {
                    NOT = {
                        any_liege_or_above = {
                            has_variable = mwp_war_scoring_owner
                            var:mwp_war_scoring_owner = scope:mwp_player
                            has_variable = mwp_war_scoring_stamp
                            var:mwp_war_scoring_stamp = scope:mwp_player.var:mwp_war_scan_stamp
                        }
                    }
                }
                scope:mwp_player = {
                    change_variable = {
                        name = mwp_war_region_root_count
                        add = 1
                    }
                }
                every_sub_realm_county = {
                    if = {
                        limit = {
                            NOT = {
                                AND = {
                                    exists = holder
                                    holder = {
                                        OR = {
                                            this = scope:mwp_player
                                            is_vassal_or_below_of = scope:mwp_player
                                        }
                                    }
                                }
                            }
                        }
                        save_scope_value_as = {
                            name = mwp_war_region_weight
                            value = 1
                        }
                        if = {
                            limit = {
                                exists = holder
                            }
                            holder = {
                                mwp_war_region_holder_weight_effect = yes
                            }
                        }
                        save_scope_value_as = {
                            name = mwp_war_county_weight
                            value = scope:mwp_war_region_weight
                        }
                        every_county_province = {
                            scope:mwp_player = {
                                change_variable = {
                                    name = mwp_war_region_barony_visits
                                    add = 1
                                }
                            }
                            mwp_war_record_county_region_effect = yes
                        }
                    }
                    else = {
                        every_county_province = {
                            scope:mwp_player = {
                                change_variable = {
                                    name = mwp_war_region_barony_visits
                                    add = 1
                                }
                            }
                        }
                    }
                }
            }
        }
    }
    every_sub_realm_county = {
        every_county_province = {
            every_neighboring_province = {
                limit = {
                    exists = county
                    NOT = {
                        mwp_war_province_in_player_subrealm_trigger = yes
                    }
                    county = {
                        exists = holder
                        holder = {
                            OR = {
                                AND = {
                                    has_variable = mwp_war_scoring_owner
                                    var:mwp_war_scoring_owner = scope:mwp_player
                                    has_variable = mwp_war_scoring_stamp
                                    var:mwp_war_scoring_stamp = scope:mwp_player.var:mwp_war_scan_stamp
                                }
                                any_liege_or_above = {
                                    has_variable = mwp_war_scoring_owner
                                    var:mwp_war_scoring_owner = scope:mwp_player
                                    has_variable = mwp_war_scoring_stamp
                                    var:mwp_war_scoring_stamp = scope:mwp_player.var:mwp_war_scan_stamp
                                }
                            }
                        }
                    }
                }
                mwp_war_mark_region_border_effect = yes
            }
        }
    }
    set_variable = {
        name = mwp_war_terrain_owner
        value = this
    }
    set_variable = {
        name = mwp_war_terrain_stamp
        value = var:mwp_war_scan_stamp
    }
}

# Caller: province. Deduplicate the same capital/border province across rivals.
mwp_war_record_fort_effect = {
    save_scope_value_as = {
        name = mwp_war_fort_level
        value = fort_level
    }
    if = {
        limit = {
            scope:mwp_war_fort_level > 0
        }
        save_scope_as = mwp_war_fort
        scope:mwp_player = {
            scope:mwp_war_fort = {
                mwp_enemy_cache_consider_fort_effect = yes
            }
            scope:mwp_war_fort = {
                mwp_war_likelihood_fort_effect = yes
            }
            change_variable = {
                name = mwp_war_fort_sample_count
                add = 1
            }
            change_variable = {
                name = mwp_war_fort_weight_sum
                add = scope:mwp_war_fort_likelihood_weight
            }
            change_variable = {
                name = mwp_war_fort_total
                add = scope:mwp_war_fort_level
            }
            if = {
                limit = {
                    scope:mwp_war_fort_level < 4
                }
                change_variable = {
                    name = mwp_war_fort_band0_count
                    add = 1
                }
                change_variable = {
                    name = mwp_war_fort_band0_weight
                    add = scope:mwp_war_fort_likelihood_weight
                }
            }
            else_if = {
                limit = {
                    scope:mwp_war_fort_level < 6
                }
                change_variable = {
                    name = mwp_war_fort_band1_count
                    add = 1
                }
                change_variable = {
                    name = mwp_war_fort_band1_weight
                    add = scope:mwp_war_fort_likelihood_weight
                }
            }
            else_if = {
                limit = {
                    scope:mwp_war_fort_level < 11
                }
                change_variable = {
                    name = mwp_war_fort_band2_count
                    add = 1
                }
                change_variable = {
                    name = mwp_war_fort_band2_weight
                    add = scope:mwp_war_fort_likelihood_weight
                }
            }
            else_if = {
                limit = {
                    scope:mwp_war_fort_level < 16
                }
                change_variable = {
                    name = mwp_war_fort_band3_count
                    add = 1
                }
                change_variable = {
                    name = mwp_war_fort_band3_weight
                    add = scope:mwp_war_fort_likelihood_weight
                }
            }
            else_if = {
                limit = {
                    scope:mwp_war_fort_level < 31
                }
                change_variable = {
                    name = mwp_war_fort_band4_count
                    add = 1
                }
                change_variable = {
                    name = mwp_war_fort_band4_weight
                    add = scope:mwp_war_fort_likelihood_weight
                }
            }
            else = {
                change_variable = {
                    name = mwp_war_fort_band5_count
                    add = 1
                }
                change_variable = {
                    name = mwp_war_fort_band5_weight
                    add = scope:mwp_war_fort_likelihood_weight
                }
            }
            if = {
                limit = {
                    var:mwp_war_fort_peak < scope:mwp_war_fort_level
                }
                set_variable = {
                    name = mwp_war_fort_peak
                    value = scope:mwp_war_fort_level
                }
            }
        }
    }
}
